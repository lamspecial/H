      authDomain:        "proj-89c9c.firebaseapp.com",
      projectId:         "proj-89c9c",

    };

    const app = initializeApp(firebaseConfig);
    const db  = collection(getFirestore(app), "messages");

    const MAX_MESSAGES = 50;

    /* تنسيق الوقت → HH:MM */
    function formatTime(ts) {
      if (!ts) return "--:--";
      const d = ts.toDate ? ts.toDate() : new Date(ts);
      return d.toTimeString().slice(0, 5);
    }

    /* رسم رسالة واحدة في الواجهة */
    const messagesEl = document.getElementById("messages");

    function renderMessage(data) {
      const line = document.createElement("div");
      line.className = "msg-line";

      const time = document.createElement("span");
      time.className = "msg-time";
      time.textContent = formatTime(data.timestamp);

      const prefix = document.createElement("span");
      prefix.className = "msg-prefix";
      prefix.textContent = "anon@chat:~$";

      const text = document.createElement("span");
      text.className = "msg-text";
      text.textContent = data.text; // textContent آمن من XSS

      line.appendChild(time);
      line.appendChild(prefix);
      line.appendChild(text);
      messagesEl.appendChild(line);
    }

    /* الاستماع الفوري — Real-Time Listener */
    const q = query(db, orderBy("timestamp", "asc"), limit(MAX_MESSAGES));

    onSnapshot(q, (snapshot) => {
      messagesEl.innerHTML = "";
      snapshot.forEach(docSnap => renderMessage(docSnap.data()));
      messagesEl.scrollTop = messagesEl.scrollHeight;
    });

    /* حذف أقدم رسالة إذا بلغ العدد الحد الأقصى */
    async function enforceLimit() {
      const snap = await getDocs(query(db, orderBy("timestamp", "asc")));
      if (snap.size >= MAX_MESSAGES) {
        await deleteDoc(doc(getFirestore(app), "messages", snap.docs[0].id));
      }
    }

    /* إرسال رسالة */
    const inputEl  = document.getElementById("msg-input");
    const sendBtn  = document.getElementById("send-btn");
    const statusEl = document.getElementById("status");

    async function sendMessage() {
      const text = inputEl.value.trim();
      if (!text) return;
      if (text.length > 500) {
        showStatus(`ERR: الرسالة تتجاوز 500 حرف (${text.length})`);
        return;
      }

      sendBtn.disabled = true;
      inputEl.disabled = true;
      showStatus("إرسال...");

      try {
        await enforceLimit();
        await addDoc(db, { text, timestamp: serverTimestamp() });
        inputEl.value = "";
        updateCounter();
        showStatus("");
      } catch (err) {
        showStatus("ERR: فشل الإرسال — " + err.code);
        console.error(err);
      } finally {
        sendBtn.disabled = false;
        inputEl.disabled = false;
        inputEl.focus();
      }
    }

    /* ربط الأحداث */
    sendBtn.addEventListener("click", sendMessage);

    inputEl.addEventListener("keydown", (e) => {
      if (e.key === "Enter" && !e.shiftKey) {
        e.preventDefault();
        sendMessage();
      }
    });

    /* عداد الأحرف */
    const counterEl = document.getElementById("char-counter");
    function updateCounter() {
      const len = inputEl.value.length;
      counterEl.textContent = `${len}/500`;
      counterEl.classList.toggle("warn", len > 400);
      counterEl.classList.toggle("over", len > 500);
    }
    inputEl.addEventListener("input", updateCounter);

    /* رسالة حالة مؤقتة */
    let statusTimer;
    function showStatus(msg) {
      statusEl.textContent = msg;
      clearTimeout(statusTimer);
      if (msg) statusTimer = setTimeout(() => { statusEl.textContent = ""; }, 3000);
    }
